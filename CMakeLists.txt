cmake_minimum_required(VERSION 3.24)
project(CatShopTakeover VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(CAT_SHOP_BUILD_SERVER "Build network service" ON)
find_package(jsoncpp CONFIG REQUIRED)
add_library(game_config_loader server/config/config_loader.cpp)
target_include_directories(game_config_loader PUBLIC server)
target_link_libraries(game_config_loader PUBLIC JsonCpp::JsonCpp)
add_library(game_core server/config/game_config.cpp server/game/logic_economy.cpp server/item/logic_item.cpp server/item/logic_item_effect.cpp server/game/logic_game.cpp server/game/logic_navigation.cpp server/game/logic_pickup.cpp server/map/grid_map.cpp server/map/map_layout.cpp server/ai/logic_cat_ai.cpp server/ai/cat_behavior.cpp server/ai/manager_behavior.cpp server/battle/logic_enemy.cpp server/battle/logic_combat.cpp server/battle/logic_progression.cpp)
target_include_directories(game_core PUBLIC server)
target_sources(game_core PRIVATE server/item/logic_random_item.cpp)
target_link_libraries(game_config_loader PUBLIC game_core)
if(MSVC)
  target_compile_options(game_core PRIVATE /W4 /utf-8)
  target_compile_options(game_config_loader PRIVATE /W4 /utf-8)
endif()
if(CAT_SHOP_BUILD_SERVER)
  find_package(Drogon CONFIG REQUIRED)
  # Drogon 1.9.13's find modules select a single .lib on multi-config MSVC.
  # Use the packages' imported targets to preserve Debug/Release selection.
  if(MSVC AND DEFINED VCPKG_TARGET_TRIPLET)
    find_package(jsoncpp CONFIG REQUIRED)
    find_package(unofficial-brotli CONFIG REQUIRED)
    set_property(TARGET Jsoncpp_lib PROPERTY INTERFACE_LINK_LIBRARIES JsonCpp::JsonCpp)
    set_property(TARGET Brotli_lib PROPERTY INTERFACE_LINK_LIBRARIES
      "unofficial::brotli::brotlidec;unofficial::brotli::brotlienc")
  endif()
  add_executable(cat_shop_server server/main.cpp server/service/game_server.cpp server/net/game_protocol.cpp server/net/game_snapshot.cpp server/net/state_stream.cpp)
  target_link_libraries(cat_shop_server PRIVATE game_core game_config_loader Drogon::Drogon)
  add_library(audio_assets server/assets/audio_library.cpp)
  target_include_directories(audio_assets PUBLIC server)
  target_link_libraries(audio_assets PUBLIC Drogon::Drogon)
  target_link_libraries(cat_shop_server PRIVATE audio_assets)
  add_executable(cat_shop_admin server/admin/main.cpp server/admin/config_repository.cpp server/admin/asset_catalog.cpp server/admin/audio_repository.cpp)
  target_link_libraries(cat_shop_admin PRIVATE audio_assets)
  target_link_libraries(cat_shop_admin PRIVATE game_config_loader Drogon::Drogon)
  if(MSVC)
    target_compile_options(audio_assets PRIVATE /W4 /utf-8)
    target_compile_definitions(audio_assets PRIVATE NOMINMAX)
    target_compile_options(cat_shop_server PRIVATE /W4 /utf-8)
    target_compile_definitions(cat_shop_server PRIVATE NOMINMAX)
    target_compile_options(cat_shop_admin PRIVATE /W4 /utf-8)
    target_compile_definitions(cat_shop_admin PRIVATE NOMINMAX)
  endif()
endif()
enable_testing()
add_executable(game_tests tests/game_tests.cpp)
target_link_libraries(game_tests PRIVATE game_core game_config_loader)
if(MSVC)
  target_compile_options(game_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME game_rules COMMAND game_tests)
add_executable(performance_tests tests/performance_tests.cpp)
target_link_libraries(performance_tests PRIVATE game_core game_config_loader)
if(MSVC)
  target_compile_options(performance_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME game_performance COMMAND performance_tests)

add_executable(config_tests tests/config_tests.cpp)
target_link_libraries(config_tests PRIVATE game_core game_config_loader)
if(MSVC)
  target_compile_options(config_tests PRIVATE /W4 /utf-8)
endif()
add_executable(ai_tests tests/ai_tests.cpp)
target_link_libraries(ai_tests PRIVATE game_core game_config_loader)
if(MSVC)
  target_compile_options(ai_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME ai_rules COMMAND ai_tests)
add_executable(manager_ai_tests tests/manager_ai_tests.cpp)
target_link_libraries(manager_ai_tests PRIVATE game_core game_config_loader)
if(MSVC)
  target_compile_options(manager_ai_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME manager_ai_rules COMMAND manager_ai_tests)
foreach(test_target game_tests performance_tests config_tests ai_tests manager_ai_tests)
  target_compile_definitions(${test_target} PRIVATE GAME_CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/data/config")
endforeach()
add_test(NAME config_rules COMMAND config_tests)

# Offline seeded matches use the same authoritative simulation as the server.
add_executable(balance_simulation tests/balance_simulation.cpp)
target_link_libraries(balance_simulation PRIVATE game_core game_config_loader)
if(MSVC)
  target_compile_options(balance_simulation PRIVATE /W4 /utf-8)
endif()
